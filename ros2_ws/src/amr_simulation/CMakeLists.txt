cmake_minimum_required(VERSION 3.10)
project(amr_simulation)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim REQUIRED)
add_library(amr_bumper_state SHARED src/bumper_state.cpp)
target_compile_features(amr_bumper_state PRIVATE cxx_std_17)
target_link_libraries(amr_bumper_state PRIVATE gz-sim::gz-sim)
install(TARGETS amr_bumper_state LIBRARY DESTINATION lib)
install(DIRECTORY worlds config DESTINATION share/${PROJECT_NAME})
ament_package()
